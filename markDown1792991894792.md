Private Sub Worksheet_Change(ByVal Target As Range)



    Dim c As Range

    Dim r As Long



    On Error GoTo SafeExit



    If Intersect(Target, Me.Columns("A")) Is Nothing Then Exit Sub



    Application.EnableEvents = False



    For Each c In Intersect(Target, Me.Columns("A"))

        r = c.Row



        If c.Value <> "" And Me.Cells(r, "D").Value = "" Then

            Me.Cells(r, "D").Value = PersianDate(Date)

        End If

    Next c



SafeExit:

    Application.EnableEvents = True



End Sub



Private Function PersianDate(ByVal dt As Date) As String



    Dim gy As Long, gm As Long, gd As Long

    Dim jy As Long, jm As Long, jd As Long

    Dim g As Long, j As Long, r As Long

    Dim leap As Boolean, i As Long

    Dim md As Variant



    gy = Year(dt)

    gm = Month(dt)

    gd = Day(dt)



    md = Array(31, 28, 31, 30, 31, 30, _

               31, 31, 30, 31, 30, 31)



    g = 365 * (gy - 1600) + (gy - 1600 + 3) \ 4 _

      - (gy - 1600 + 99) \ 100 _

      + (gy - 1600 + 399) \ 400



    For i = 0 To gm - 2

        g = g + md(i)

    Next i



    leap = ((gy Mod 4 = 0 And gy Mod 100 <> 0) Or gy Mod 400 = 0)



    If gm > 2 And leap Then g = g + 1



    g = g + gd - 1

    j = g - 79



    jy = 979 + 33 * (j \ 12053)

    j = j Mod 12053



    jy = jy + 4 * (j \ 1461)

    j = j Mod 1461



    If j >= 366 Then

        jy = jy + (j - 1) \ 365

        j = (j - 1) Mod 365

    End If



    If j < 186 Then

        jm = 1 + j \ 31

        jd = 1 + j Mod 31

    Else

        jm = 7 + (j - 186) \ 30

        jd = 1 + (j - 186) Mod 30

    End If



    PersianDate = Format(jy, "0000") & "/" & _

                  Format(jm, "00") & "/" & Format(jd, "00")



End Function
